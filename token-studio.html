async function saveProject(){ const project = collectProject(); if (!validateProject(project)){ return; } const button = $("#saveButton"); const original = button.textContent; button.disabled = true; button.textContent = "Saving..."; try { let savedRemotely = false; if (supabase){ try { const user = await getUser(); if (user){ /* * Supabase public.projects * * Saves: * - project_name * - token_name * - token_symbol * - decimals * - total_supply * - network * - description * - freeze_authority * - mint_authority * - metadata_enabled * - token_status */ const result = await supabase .from("projects") .insert({ user_id: user.id, project_name: project.project_name, token_name: project.token_name, token_symbol: project.token_symbol, decimals: project.decimals, total_supply: project.total_supply, network: project.network, description: project.description, freeze_authority: project.freeze_authority, mint_authority: project.mint_authority, metadata_enabled: project.metadata, token_status: "draft" }) .select() .single(); if (!result.error){ savedRemotely = true; console.log( "[Sonativa] Project saved:", result.data ); } else { console.error( "[Sonativa] Supabase project insert error:", result.error ); showToast( "Supabase error", result.error.message || "Unable to save project to Supabase.", "error" ); } } else { console.error( "[Sonativa] No authenticated user." ); showToast( "Authentication required", "Please sign in before saving your project.", "error" ); } } catch (remoteError){ console.error( "[Sonativa] Remote project save error:", remoteError ); showToast( "Supabase error", remoteError?.message || "Unable to connect to Supabase.", "error" ); } } /* * IMPORTANT: * We do NOT silently save locally anymore when Supabase * is available but rejects the project. * * This prevents the user from thinking the project was * saved to Supabase when it was actually only saved locally. */ if (!savedRemotely){ return; } showToast( "Project saved", "Your project draft was saved to your Sonativa workspace.", "success" ); setTimeout(() => { window.location.href = "projects.html"; }, 1000); } catch(error){ console.error( "[Sonativa] Save project error:", error ); showToast( "Save failed", error?.message || "Unable to save this project.", "error" ); } finally { button.disabled = false; button.textContent = original; } }
